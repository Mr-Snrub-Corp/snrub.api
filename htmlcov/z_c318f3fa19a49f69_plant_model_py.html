1"""True-state plant integrator (Phase 3, docs/roadmap.md). 

2 

3Integrates a persistent physical state toward incident-derived targets 

4(services/telemetry.compute_targets) with metric-to-metric coupling and one 

5negative temperature -> reactivity feedback term. Deterministic — measurement 

6noise belongs to the sensor layer (services/sensors.py). 

7 

8Stability: each metric is a first-order lerp with the exact discretization 

9``alpha = 1 - exp(-dt / tau)``, so alpha is always in (0, 1) — unconditionally 

10stable and overshoot-free for any dt. The coupling graph is a DAG 

11(flow -> temp, power -> temp, reactivity -> power, containment -> radiation) 

12plus one deliberate cycle temp -> reactivity -> power -> temp whose loop gain 

13is ``-K_TEMP_REACTIVITY * K_REACTIVITY_POWER * K_POWER_TEMP = -0.0375``; 

14|gain| << 1 and negative, so the system has a unique fixed point near the 

15targets and converges monotonically at dt = 1s. With zero active incidents 

16every coupling term vanishes and the fixed point is exactly BASE_METRICS. 

17""" 

18

53# Time constants (seconds): how fast each metric eases toward its target. 

54# Gameplay-feel knobs — fast neutronics, slow thermal mass and structure. 

55TAUS: dict[str, float] = { 

56 "reactivity": 5.0, 

57 "reactor_power": 8.0, 

58 "coolant_flow_rate": 10.0, 

59 "coolant_pressure": 12.0, 

60 "radiation_level": 15.0, 

61 "core_temperature": 30.0,

65# Coupling gains, seeded from the cross-metric ratios in INCIDENT_IMPACT_MAP 

66# (mean delta ratio of the incident codes touching both metrics). 

67K_FLOW_TEMP = 4.0 # °C per % coolant-flow deficit 

68K_POWER_TEMP = 2.5 # °C per % reactor-power excess 

69K_REACTIVITY_POWER = 3.0 # % power per reactivity unit 

70K_CONTAINMENT_RADIATION = 5.0 # mSv/h per % containment deficit 

71K_TEMP_REACTIVITY = 0.005 # reactivity per °C above base (Doppler-style negative feedback)

73# --- Actuators (Phase 4, docs/roadmap.md) ------------------------------------- 

74# God-mode levers set these; targets_from_actuators folds them into the same 

75# target shape compute_targets produced, so step() integrates identically. 

76# Commands (rod/pump/steam) set the operating point; faults (leak/xenon) push 

77# away from it. Magnitudes mirror the incident deltas they replace in 

78# telemetry.INCIDENT_IMPACT_MAP (primary_coolant_loss, xenon_poisoning). 

79ACTUATOR_NOMINAL: dict[str, float] = { 

80 "rod_position": 50.0, # % inserted; 50 = critical (reactivity 0) 

81 "pump_speed": 80.0, # % ; drives coolant flow, nominal = base flow 

82 "steam_valve": 50.0, # % open; 50 = base pressure 

83 "leak_rate": 0.0, # fault; 0 = no leak 

84 "xenon_injection": 0.0, # fault; 0 = no poisoning 

85} 

86 

87ACTUATOR_RANGES: dict[str, tuple[float, float]] = dict.fromkeys(ACTUATOR_NOMINAL, (0.0, 100.0)) 

88 

89K_ROD_REACTIVITY = 0.1 # reactivity per % rod displaced from nominal -> ±5 at the rails 

90K_VALVE_PRESSURE = 1.4 # bar per % steam-valve displaced from nominal -> ±70 

91K_LEAK_FLOW = 0.2 # coolant-flow %-pts lost per leak unit -> -20 at leak 100 

92K_LEAK_TEMP = 1.5 # °C per leak unit -> +150 at leak 100 

93K_XENON_REACTIVITY = 0.03 # reactivity suppressed per xenon unit -> -3 at xenon 100

96def initial_state() -> dict[str, float]: 

97 """Quiescent plant: every metric at its base value.""" 

98 return dict(BASE_METRICS)

101def _clamp(value: float, low: float, high: float) -> float: 

102 return max(low, min(high, value))

105def _effective_targets(state: dict[str, float], targets: dict[str, float]) -> dict[str, float]: 

106 """Incident targets adjusted by the current state's coupling terms.""" 

107 eff = dict(targets) 

108 eff["core_temperature"] += K_FLOW_TEMP * (BASE_COOLANT_FLOW_RATE - state["coolant_flow_rate"]) + K_POWER_TEMP * ( 

109 state["reactor_power"] - BASE_REACTOR_POWER_OUTPUT 

110 ) 

111 eff["reactor_power"] += K_REACTIVITY_POWER * state["reactivity"] 

112 eff["radiation_level"] += K_CONTAINMENT_RADIATION * (BASE_CONTAINMENT_INTEGRITY - state["containment_integrity"]) 

113 eff["reactivity"] -= K_TEMP_REACTIVITY * (state["core_temperature"] - BASE_CORE_TEMPERATURE) 

114 return eff

117def step(state: dict[str, float], targets: dict[str, float], elapsed_seconds: float) -> dict[str, float]: 

118 """Advance true plant state toward ``targets`` by ``elapsed_seconds``. 

119 

120 Inputs are not mutated. Each metric eases a fraction of the remaining gap 

121 (larger ``elapsed_seconds`` → closer to the target). The simulator passes 

122 1.0 (one tick). 

123 

124 Returns the new true state, same keys as METRICS, e.g. after one second 

125 at the base targets:: 

126 

127 {"reactor_power": 95.0, "core_temperature": 700.0, "reactivity": 0.0, 

128 "coolant_flow_rate": 80.0, "coolant_pressure": 130.0, 

129 "radiation_level": 2.0, "containment_integrity": 95.0} 

130 """ 

131 eff = _effective_targets(state, targets) 

132 result: dict[str, float] = {} 

133 for metric in METRICS: 

134 alpha = 1.0 - exp(-elapsed_seconds / TAUS[metric]) 

135 value = state[metric] + (eff[metric] - state[metric]) * alpha 

136 result[metric] = _clamp(value, *PHYSICAL_RANGES[metric]) 

137 return result

140def targets_from_actuators(actuators: dict[str, float]) -> dict[str, float]: 

141 """Fold actuator/fault positions into plant targets (Phase 4, docs/roadmap.md). 

142 

143 Replaces compute_targets(incidents) as the plant driver: commands set the 

144 operating point, faults push away from it. Missing actuators fall back to 

145 their nominal position, so an empty dict yields exactly BASE_METRICS. Only 

146 sets targets — coupling and clamping still happen in step(). 

147 """ 

148 

149 def _value(name: str) -> float: 

150 return actuators.get(name, ACTUATOR_NOMINAL[name]) 

151 

152 rod = _value("rod_position") 

153 pump = _value("pump_speed") 

154 valve = _value("steam_valve") 

155 leak = _value("leak_rate") 

156 xenon = _value("xenon_injection") 

157 

158 targets = dict(BASE_METRICS) 

159 # Commands — normal operating point. 

160 targets["reactivity"] = BASE_REACTIVITY + K_ROD_REACTIVITY * (ACTUATOR_NOMINAL["rod_position"] - rod) 

161 targets["coolant_flow_rate"] = pump 

162 targets["coolant_pressure"] = BASE_COOLANT_PRESSURE + K_VALVE_PRESSURE * (ACTUATOR_NOMINAL["steam_valve"] - valve) 

163 # Faults — malfunctions layered on top of the commanded point. 

164 targets["coolant_flow_rate"] -= K_LEAK_FLOW * leak 

165 targets["core_temperature"] = BASE_CORE_TEMPERATURE + K_LEAK_TEMP * leak 

166 targets["reactivity"] -= K_XENON_REACTIVITY * xenon 

167 return targets
